import fs from 'node:fs';
import {fileURLToPath} from 'node:url';
import path from 'node:path';
const args=process.argv.slice(2),terms=[],filters={};
for(let i=0;i<args.length;i++){if(args[i].startsWith('--')){const name=args[i].slice(2);if(!['source','kind','license','limit'].includes(name)||!args[i+1])throw Error('Use search terms and optional --source, --kind, --license, --limit.');filters[name]=args[++i];}else terms.push(args[i].toLowerCase());}
const catalog=JSON.parse(fs.readFileSync(new URL('./catalog.json',import.meta.url)));
const found=catalog.components.filter(item=>{
 if(filters.source&&!item.source_slug.includes(filters.source))return false;
 if(filters.kind&&item.kind!==filters.kind)return false;
 if(filters.license&&item.source_license?.spdx!==filters.license)return false;
 const text=[item.name,item.category,...(item.tags||[]),...(item.frameworks||[]),item.source_slug,item.source_path,item.docs_url].filter(Boolean).join(' ').toLowerCase();
 return terms.every(term=>text.includes(term));
});
const limit=filters.limit===undefined?found.length:Number(filters.limit);if(!Number.isSafeInteger(limit)||limit<0)throw Error('Use a nonnegative limit.');
console.log(JSON.stringify({query:terms,matched:found.length,returned:Math.min(limit,found.length),items:found.slice(0,limit)},null,2));
